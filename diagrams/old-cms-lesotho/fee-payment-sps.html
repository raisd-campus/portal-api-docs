  participant REG as qrcode_gen.php
  participant QR as qr.limkokwing.net
  participant VER as verification.limkokwing.net
  participant CMS as campus2_cambodia

  QR-->>REG: PNG (encodes /qrcode/slug)
  actor Public as Scanner
  Public->>QR: GET /qrcode/slug

  VER->>CMS: CALL GetStudentCertInfo
  CMS-->>VER: cert metadata + blob
